rules_version = '2';
service cloud.firestore {
 match /databases/{database}/documents {
  function signedIn(){return request.auth != null;}
  function isRegistrarFor(schoolId){return signedIn() && exists(/databases/$(database)/documents/registrars/$(request.auth.uid)) && get(/databases/$(database)/documents/registrars/$(request.auth.uid)).data.schoolId == schoolId;}
  function isApprovedParent(){return signedIn() && exists(/databases/$(database)/documents/parents/$(request.auth.uid)) && get(/databases/$(database)/documents/parents/$(request.auth.uid)).data.status == 'approved';}
  function parentOwnsStudent(schoolId,studentId){return isApprovedParent() && get(/databases/$(database)/documents/parents/$(request.auth.uid)).data.schoolId == schoolId && studentId in get(/databases/$(database)/documents/parents/$(request.auth.uid)).data.matchedStudentIds;}
  match /parents/{parentUid} {
   allow create: if signedIn() && request.auth.uid == parentUid && request.resource.data.role == 'parent' && request.resource.data.status == 'pending' && request.resource.data.matchedStudentIds is list;
   allow read: if signedIn() && (request.auth.uid == parentUid || isRegistrarFor(resource.data.schoolId));
   allow update: if signedIn() && isRegistrarFor(resource.data.schoolId) && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status','reviewedAt','reviewedBy','updatedAt']);
   allow delete: if false;
  }
  match /schools/{schoolId}/students/{studentId} {
   allow read: if parentOwnsStudent(schoolId,studentId) || isRegistrarFor(schoolId) || (signedIn() && exists(/databases/$(database)/documents/schoolTeachers/$(schoolId + '_' + request.auth.uid)) && get(/databases/$(database)/documents/schoolTeachers/$(schoolId + '_' + request.auth.uid)).data.status == 'connected');
   // Preserve your existing school/teacher write rules here; do not replace them blindly.
  }
 }
}
